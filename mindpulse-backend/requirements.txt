# MindPulse backend requirements
#
# Python 3.12 is required. The emotion model artifacts in emotion_model/ were
# saved with the exact versions pinned below (TensorFlow/Keras for the .h5
# model, scikit-learn for the .joblib files). These are pinned on purpose:
# an unpinned install can load the model with a different Keras/sklearn and
# either fail to load or produce different predictions.
#
# Install with:  python -m pip install -r requirements.txt
# (first install downloads ~700 MB of TensorFlow/MediaPipe - allow a few minutes)

# --- Web framework ---
Django==5.2.7
djangorestframework==3.18.0
django-cors-headers==4.9.0
python-dotenv==1.2.1
supabase==2.31.0

# --- Emotion model inference (emotion_model/ package, transferred from capstone) ---
tensorflow==2.21.0
keras==3.15.0
mediapipe==0.10.35
# MediaPipe itself depends on opencv-contrib-python, so that is pinned here
# instead of opencv-python (pinning a different OpenCV build would install a
# second, conflicting cv2).
opencv-contrib-python==5.0.0.93
scikit-learn==1.9.0
joblib==1.5.3
numpy==2.5.1
